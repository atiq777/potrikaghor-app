@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun MyPapersScreen(
    appRepo: AppStateRepository,
    onOpenReader: (Paper, String) -> Unit,
    onAddPaper: () -> Unit,
    onFixUrl: (Paper) -> Unit
) {
    val appState by appRepo.state.collectAsState()
    val reducedMotion = rememberReducedMotion()
    
    var selectedCat by remember { mutableStateOf("all") }
    var selectedDivision by remember { mutableStateOf("all") }
    var searchQuery by remember { mutableStateOf("") }


    val allList = appRepo.allPapers
    val directoryEntries = remember { DirectoryRepository.getAllEntries() }
    val divisionByUrl = remember(directoryEntries) {
        directoryEntries.associate { it.websiteUrl to it.division }
    }


    val combinedList = remember(allList, directoryEntries, appState.onlyShowFavorites, appState.hiddenPapers, appState.urlFixes) {
        val customPapers = allList.filter { it.custom }
        val mappedDirectory = directoryEntries
            .filter { !appState.hiddenPapers.contains(it.websiteUrl) }
            .map { dir ->
                val normalizedCat = when (dir.primaryCategory) {
                    "bu" -> "biz"
                    "on" -> "online"
                    "te" -> "tech"
                    "ed" -> "edu"
                    "sp" -> "sports"
                    "jo" -> "jobs"
                    "ko" -> "kol"
                    "lo" -> "local"
                    "ma" -> "mag"
                    "bl" -> "blog"
                    else -> dir.primaryCategory
                }
                Paper(
                    id = dir.directoryId,
                    cat = normalizedCat,
                    bn = dir.newspaperName,
                    en = dir.newspaperName,
                    url = dir.websiteUrl,
                    custom = false,
                    fixedUrl = appState.urlFixes[dir.websiteUrl]
                )
            }
        val merged = customPapers + mappedDirectory
        if (appState.onlyShowFavorites) merged.filter { appState.favs.contains(it.url) } else merged
    }


    val filteredList = remember(combinedList, selectedCat, selectedDivision, searchQuery, appState.favs) {
        combinedList.filter { p ->
            val matchCat = when (selectedCat) {
                "all" -> true
                "mine" -> p.custom
                "fav" -> appState.favs.contains(p.url)
                else -> p.cat == selectedCat
            }
            val matchDiv = if (selectedCat != "local" || selectedDivision == "all") {
                true
            } else {
                divisionByUrl[p.url] == selectedDivision
            }
            val matchQuery = if (searchQuery.trim().isEmpty()) true else {
                val q = searchQuery.trim().lowercase()
                p.bn.lowercase().contains(q) || p.en.lowercase().contains(q) || p.url.lowercase().contains(q)
            }
            matchCat && matchDiv && matchQuery
        }
    }


    val viewModeInteraction = remember { MutableInteractionSource() }
    val fabInteraction = remember { MutableInteractionSource() }


    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("আমার পত্রিকা", fontFamily = FontFamily.Serif, fontWeight = FontWeight.Bold) },
                actions = {
                    IconButton(
                        onClick = { appRepo.toggleGridView() },
                        interactionSource = viewModeInteraction,
                        modifier = Modifier.pressScaleEffect(viewModeInteraction)
                    ) {
                        Icon(
                            imageVector = if (appState.gridView) Icons.Filled.ViewAgenda else Icons.Filled.GridView,
                            contentDescription = "ভিউ পরিবর্তন করুন"
                        )
                    }
                }
            )
        },
        floatingActionButton = {
            FloatingActionButton(
                onClick = onAddPaper,
                interactionSource = fabInteraction,
                modifier = Modifier.pressScaleEffect(fabInteraction, pressedScale = 0.9f)
            ) {
                Icon(Icons.Filled.Add, contentDescription = "নতুন পত্রিকা যোগ করুন")
            }
        }
    ) { paddingValues ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(paddingValues)
        ) {
            // Search Bar
            OutlinedTextField(
                value = searchQuery,
                onValueChange = { searchQuery = it },
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 14.dp, vertical = 4.dp),
                placeholder = { Text("পত্রিকার নাম বা লিংক খুঁজুন...", fontSize = 13.5.sp) },
                leadingIcon = { Icon(Icons.Filled.Search, contentDescription = null) },
                trailingIcon = {
                    if (searchQuery.isNotEmpty()) {
                        IconButton(onClick = { searchQuery = "" }) {
                            Icon(Icons.Filled.Clear, contentDescription = "মুছুন")
                        }
                    }
                },
                singleLine = true,
                shape = RoundedCornerShape(14.dp)
            )


            // Category Chips Row
            val availableCategories = listOf(
                Pair("all", "সব"),
                Pair("fav", "প্রিয়"),
                Pair("mine", "নিজের যোগ করা")
            ) + PapersData.categories.drop(1).map { Pair(it.id, it.label) }


            LazyRow(
                contentPadding = PaddingValues(horizontal = 14.dp, vertical = 4.dp),
                horizontalArrangement = Arrangement.spacedBy(7.dp)
            ) {
                items(availableCategories, key = { it.first }) { (id, label) ->
                    val isSelected = selectedCat == id
                    val chipInteraction = remember { MutableInteractionSource() }
                    val chipBg by animateColorAsState(
                        targetValue = if (isSelected) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.surfaceVariant,
                        animationSpec = tween(
                            durationMillis = if (reducedMotion) 0 else MotionTokens.NORMAL_MS,
                            easing = FastOutSlowInEasing
                        ),
                        label = "paperCatChipBg"
                    )
                    val chipText by animateColorAsState(
                        targetValue = if (isSelected) MaterialTheme.colorScheme.onPrimary else MaterialTheme.colorScheme.onSurfaceVariant,
                        animationSpec = tween(
                            durationMillis = if (reducedMotion) 0 else MotionTokens.FAST_MS,
                            easing = FastOutSlowInEasing
                        ),
                        label = "paperCatChipText"
                    )


                    FilterChip(
                        selected = isSelected,
                        onClick = { 
                            selectedCat = id 
                            if(id != "local") {
                                selectedDivision = "all"
                            }
                        },
                        interactionSource = chipInteraction,
                        modifier = Modifier.pressScaleEffect(chipInteraction, pressedScale = 0.95f),
                        label = {
                            Text(
                                text = label,
                                fontSize = 12.5.sp,
                                fontWeight = if (isSelected) FontWeight.Bold else FontWeight.Medium
                            )
                        },
                        colors = FilterChipDefaults.filterChipColors(
                            selectedContainerColor = chipBg,
                            selectedLabelColor = chipText,
                            containerColor = chipBg,
                            labelColor = chipText
                        ),
                        shape = RoundedCornerShape(16.dp),
                        border = null
                    )
                }
            }


            // Division Filter Row (Only show if Local category is selected)
            if (selectedCat == "local") {
                val divisions = listOf(
                    "all" to "সব বিভাগ", 
                    "Dhaka" to "ঢাকা", 
                    "Chittagong" to "চট্টগ্রাম", 
                    "Rajshahi" to "রাজশাহী", 
                    "Khulna" to "খুলনা", 
                    "Barisal" to "বরিশাল", 
                    "Sylhet" to "সিলেট", 
                    "Rangpur" to "রংপুর", 
                    "Mymensingh" to "ময়মনসিংহ"
                )
                LazyRow(
                    contentPadding = PaddingValues(horizontal = 14.dp, vertical = 2.dp),
                    horizontalArrangement = Arrangement.spacedBy(7.dp)
                ) {
                    items(divisions, key = { it.first }) { (id, label) ->
                        FilterChip(
                            selected = selectedDivision == id,
                            onClick = { selectedDivision = id },
                            label = { Text(label, fontSize = 11.5.sp) },
                            shape = RoundedCornerShape(14.dp)
                        )
                    }
                }
            }


            // Only favorites toggle row
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 14.dp, vertical = 2.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Text(
                    text = "শুধু পছন্দের পত্রিকা প্রদর্শন",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
                Switch(
                    checked = appState.onlyShowFavorites,
                    onCheckedChange = { appRepo.setOnlyShowFavorites(it) }
                )
            }


            val isAppReady by appRepo.isReady.collectAsState()
            val bodyState = when {
                !isAppReady -> "loading"
                filteredList.isEmpty() -> "empty"
                appState.gridView -> "grid"
                else -> "list"
            }


            Crossfade(
                targetState = bodyState,
                animationSpec = tween(
                    durationMillis = if (reducedMotion) 0 else MotionTokens.NORMAL_MS,
                    easing = FastOutSlowInEasing
                ),
                label = "myPapersBodyCrossfade"
            ) { state ->
                when (state) {
                    "loading" -> {
                        com.potrikaghor.ui.components.BrandedLoadingIndicator(
                            message = "পত্রিকা প্রস্তুত হচ্ছে...",
                            logoSize = 72.dp,
                            modifier = Modifier.fillMaxSize()
                        )
                    }
                    "empty" -> {
                        EmptyState(
                            icon = Icons.Filled.Star,
                            title = "কোনো পত্রিকা মেলেনি",
                            body = "আপনার অনুসন্ধান বা নির্বাচিত ক্যাটাগরিতে কোনো পত্রিকা পাওয়া যায়নি।",
                            actionLabel = "সব দেখুন",
                            onAction = {
                                selectedCat = "all"
                                selectedDivision = "all"
                                searchQuery = ""
                                appRepo.setOnlyShowFavorites(false)
                            }
                        )
                    }
                    "grid" -> {
                        LazyVerticalGrid(
                            columns = GridCells.Fixed(2),
                            contentPadding = PaddingValues(start = 14.dp, end = 14.dp, bottom = 82.dp, top = 6.dp),
                            horizontalArrangement = Arrangement.spacedBy(8.dp),
                            verticalArrangement = Arrangement.spacedBy(8.dp)
                        ) {
                            items(filteredList, key = { it.url }) { paper ->
                                MyPaperGridItem(
                                    paper = paper,
                                    isFav = appState.favs.contains(paper.url),
                                    onToggleFav = { appRepo.toggleFav(paper.url) },
                                    onOpen = { onOpenReader(paper, paper.openUrl) },
                                    onFixUrl = { onFixUrl(paper) }
                                )
                            }
                        }
                    }
                    else -> {
                        LazyColumn(
                            contentPadding = PaddingValues(bottom = 82.dp, top = 4.dp)
                        ) {
                            items(filteredList, key = { it.url }) { paper ->
                                MyPaperListItem(
                                    paper = paper,
                                    isFav = appState.favs.contains(paper.url),
                                    onToggleFav = { appRepo.toggleFav(paper.url) },
                                    onOpen = { onOpenReader(paper, paper.openUrl) },
                                    onFixUrl = { onFixUrl(paper) },
                                    onToggleHide = { appRepo.toggleHidePaper(paper.url) }
                                )
                            }
                        }
                    }
                }
            }
        }
    }
}


@Composable
fun MyPaperListItem(
    paper: Paper,
    isFav: Boolean,
    onToggleFav: () -> Unit,
    onOpen: () -> Unit,
    onFixUrl: () -> Unit,
    onToggleHide: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }


    Card(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 14.dp, vertical = 6.dp)
            .subtleClickable(interactionSource, onClick = onOpen),
        shape = RoundedCornerShape(12.dp),
        colors = CardDefaults.cardColors(
            containerColor = MaterialTheme.colorScheme.surface
        )
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(12.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            PaperAvatar(
                name = paper.bn,
                isCustom = paper.custom,
                url = paper.openUrl,
                size = 44.dp
            )


            Spacer(modifier = Modifier.width(14.dp))


            Column(modifier = Modifier.weight(1f)) {
                Text(
                    text = paper.bn,
                    fontSize = 15.5.sp,
                    fontWeight = FontWeight.Bold,
                    fontFamily = FontFamily.Serif,
                    color = MaterialTheme.colorScheme.onSurface,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis
                )
                Spacer(modifier = Modifier.height(2.dp))
                Text(
                    text = paper.openUrl,
                    fontSize = 11.5.sp,
                    color = MaterialTheme.colorScheme.primary,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis
                )
            }


            Spacer(modifier = Modifier.width(8.dp))


            if (paper.custom) {
                IconButton(onClick = onFixUrl, modifier = Modifier.size(36.dp)) {
                    Icon(
                        imageVector = Icons.Filled.Edit,
                        contentDescription = "লিংক সম্পাদনা করুন",
                        tint = MaterialTheme.colorScheme.onSurfaceVariant,
                        modifier = Modifier.size(18.dp)
                    )
                }
            } else {
                IconButton(onClick = onToggleHide, modifier = Modifier.size(36.dp)) {
                    Icon(
                        imageVector = Icons.Filled.Clear,
                        contentDescription = "লুকান",
                        tint = MaterialTheme.colorScheme.onSurfaceVariant,
                        modifier = Modifier.size(18.dp)
                    )
                }
            }


            IconButton(onClick = onToggleFav, modifier = Modifier.size(36.dp)) {
                Icon(
                    imageVector = if (isFav) Icons.Filled.Star else Icons.Outlined.StarBorder,
                    contentDescription = if (isFav) "প্রিয় থেকে সরান" else "প্রিয়তে যোগ করুন",
                    tint = if (isFav) FlagRed else MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.size(22.dp)
                )
            }
        }
    }
}


@Composable
fun MyPaperGridItem(
    paper: Paper,
    isFav: Boolean,
    onToggleFav: () -> Unit,
    onOpen: () -> Unit,
    onFixUrl: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }


    Card(
        modifier = Modifier
            .fillMaxWidth()
            .subtleClickable(interactionSource, onClick = onOpen),
        shape = RoundedCornerShape(14.dp),
        colors = CardDefaults.cardColors(
            containerColor = MaterialTheme.colorScheme.surface
        )
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(12.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.Top
            ) {
                if (paper.custom) {
                    IconButton(onClick = onFixUrl, modifier = Modifier.size(32.dp)) {
                        Icon(
                            imageVector = Icons.Filled.Edit,
                            contentDescription = "লিংক সম্পাদনা",
                            tint = MaterialTheme.colorScheme.onSurfaceVariant,
                            modifier = Modifier.size(16.dp)
                        )
                    }
                } else {
                    Spacer(modifier = Modifier.size(32.dp))
                }


                IconButton(onClick = onToggleFav, modifier = Modifier.size(32.dp)) {
                    Icon(
                        imageVector = if (isFav) Icons.Filled.Star else Icons.Outlined.StarBorder,
                        contentDescription = "প্রিয়",
                        tint = if (isFav) FlagRed else MaterialTheme.colorScheme.onSurfaceVariant,
                        modifier = Modifier.size(20.dp)
                    )
                }
            }


            PaperAvatar(
                name = paper.bn,
                isCustom = paper.custom,
                url = paper.openUrl,
                size = 56.dp
            )


            Spacer(modifier = Modifier.height(10.dp))


            Text(
                text = paper.bn,
                fontSize = 14.5.sp,
                fontWeight = FontWeight.Bold,
                fontFamily = FontFamily.Serif,
                color = MaterialTheme.colorScheme.onSurface,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )


            Spacer(modifier = Modifier.height(2.dp))


            Text(
                text = paper.openUrl.replace("https://", "").replace("www.", ""),
                fontSize = 11.sp,
                color = MaterialTheme.colorScheme.primary,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
    }
